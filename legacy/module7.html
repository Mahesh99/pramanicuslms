Section 7.1

Importing Modules

A module is a Python file (.py) containing functions, classes, and variables. Python's standard library contains hundreds of pre-written modules. You bring them into your code with import.

Import styles
# Style 1: import the whole module
import math
print(math.pi)           # access with module.name
print(math.sqrt(16))

# Style 2: import specific names
from math import pi, sqrt, floor
print(pi)                # use directly without prefix
print(sqrt(25))

# Style 3: import with alias
import datetime as dt
from math import sqrt as square_root

# Style 4: import everything (NOT recommended)
from math import *   # pollutes namespace — avoid in production code

# if __name__ == "__main__" — the entry point guard
# This block runs only when the file is run directly,
# NOT when it is imported as a module
if __name__ == "__main__":
    print("Running as main script")
💡
Import order (PEP 8): 1) Standard library imports, 2) Third-party library imports, 3) Local module imports. Each group separated by a blank line.

Section 7.2

Creating Your Own Module

Any .py file is a module. Create one to organise reusable code.

mathutils.py — your own module
# mathutils.py
"""Utility functions for mathematical operations."""

PI = 3.14159265358979

def circle_area(r):
    """Return the area of a circle with radius r."""
    return PI * r * r

def is_prime(n):
    """Return True if n is a prime number."""
    if n < 2: return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0: return False
    return True

# This only runs when mathutils.py is run directly
if __name__ == "__main__":
    print(circle_area(5))   # test
main.py — using your module
# main.py (same directory as mathutils.py)
from mathutils import circle_area, is_prime

print(circle_area(7))    # 153.94...
print(is_prime(17))      # True
print(is_prime(18))      # False

Section 7.3

Packages

A package is a directory containing multiple modules and a special __init__.py file. It lets you organise a large project into a namespace hierarchy.

Package structure
my_project/
│
├── main.py
└── utils/               # ← this is a package
    ├── __init__.py      # marks directory as a package
    ├── math_utils.py
    ├── string_utils.py
    └── file_utils.py
Importing from packages
# In main.py:
from utils import math_utils           # import whole module
from utils.math_utils import is_prime  # import specific function
import utils.string_utils as su       # with alias

# __init__.py can expose a cleaner API:
# utils/__init__.py
from .math_utils   import is_prime, circle_area
from .string_utils import slugify

# Then in main.py:
from utils import is_prime   # feels like it's directly in 'utils'

Section 7.4

Virtual Environments & pip

A virtual environment is an isolated Python installation for one project. This prevents conflicting dependencies between projects. Always use one.

Virtual environment workflow
# 1. Create a virtual environment in your project folder
python -m venv venv

# 2. Activate it
source venv/bin/activate        # Linux / macOS
venv\Scripts\activate           # Windows CMD
venv\Scripts\Activate.ps1       # Windows PowerShell

# Your terminal prompt should now show (venv)

# 3. Install packages with pip
pip install requests             # install latest version
pip install requests==2.31.0     # install specific version
pip install requests pandas      # install multiple

# 4. See installed packages
pip list
pip show requests

# 5. Save your dependencies to a file
pip freeze > requirements.txt

# 6. Another developer installs from your file
pip install -r requirements.txt

# 7. Deactivate when done
deactivate
✅
Every project = its own virtual environment. Never install packages globally (without a venv). Commit requirements.txt to git so others can recreate the exact environment.

Section 7.5

The datetime Module

datetime module
from datetime import datetime, date, timedelta

# Current date and time
now   = datetime.now()
today = date.today()
print(now)           # 2026-07-04 09:30:00.123456
print(today)         # 2026-07-04

# Create a specific date
birthday = date(2000, 6, 15)
dt       = datetime(2026, 7, 4, 9, 30)

# Access components
print(now.year, now.month, now.day)
print(now.hour, now.minute, now.second)

# Format: datetime → string
print(now.strftime("%d-%m-%Y"))       # 04-07-2026
print(now.strftime("%I:%M %p"))       # 09:30 AM
print(now.strftime("%A, %B %d %Y"))   # Saturday, July 04 2026

# Parse: string → datetime
parsed = datetime.strptime("04/07/2026", "%d/%m/%Y")

# Date arithmetic
one_week  = timedelta(days=7)
next_week = today + one_week
print(next_week)

age_days = (today - birthday).days
print(f"Age: {age_days // 365} years")

Section 7.6

The math Module

math module
import math

# Constants
math.pi       # 3.14159265...
math.e        # 2.71828...  (Euler's number)
math.tau      # 2π = 6.28318...
math.inf      # infinity

# Rounding
math.ceil(3.2)    # 4   — round up
math.floor(3.9)  # 3   — round down
math.trunc(3.9)  # 3   — truncate (same as int())

# Powers and roots
math.sqrt(25)         # 5.0  — square root
math.pow(2, 10)       # 1024.0 (float)
math.cbrt(27)         # 3.0  — cube root (Python 3.11+)
math.log(100, 10)     # 2.0  — log base 10
math.log2(8)          # 3.0
math.log(math.e)      # 1.0  — natural log

# Combinatorics
math.factorial(5)    # 120
math.comb(10, 3)     # 120  — combinations: C(10,3)
math.perm(10, 3)     # 720  — permutations: P(10,3)
math.gcd(48, 18)     # 6    — greatest common divisor
math.lcm(4, 6)       # 12   — least common multiple (3.9+)

# Trigonometry (angles in radians)
math.sin(math.pi / 2)  # 1.0
math.cos(0)            # 1.0
math.degrees(math.pi)  # 180.0  — radians → degrees
math.radians(180)      # 3.14159... — degrees → radians

# Hypot — Euclidean distance (Pythagoras)
math.hypot(3, 4)   # 5.0 (√(3²+4²))

Section 7.7

The random Module

random module
import random

# Random float 0.0 ≤ x < 1.0
random.random()             # 0.742...

# Random integer (inclusive on both ends)
random.randint(1, 100)      # e.g. 42

# Random float in a range
random.uniform(1.0, 5.0)   # e.g. 3.72

# Random choice from a sequence
fruits = ["apple", "banana", "cherry"]
random.choice(fruits)        # "banana"

# Multiple unique choices (no replacement)
random.sample(fruits, k=2)   # ["cherry", "apple"]

# Multiple choices (with replacement)
random.choices(fruits, k=5, weights=[3,1,1])  # weighted random

# Shuffle a list IN PLACE
deck = list(range(1, 53))
random.shuffle(deck)         # modifies deck directly

# Set a seed for reproducibility (useful in testing/exams)
random.seed(42)
print(random.randint(1, 100))  # always same result with same seed
⚠️
Not cryptographically secure. Never use random for passwords, tokens, or security-sensitive values. Use secrets module instead: import secrets; secrets.randbelow(100).

os & sys Modules

os — Operating System Interface

os module
import os

# Working directory
os.getcwd()                     # current working directory
os.chdir("/path/to/dir")         # change directory

# Listing and file operations
os.listdir(".")                  # list files in current dir
os.makedirs("output/data", exist_ok=True)  # create dirs
os.remove("file.txt")            # delete a file
os.rename("old.txt", "new.txt") # rename

# Path operations (prefer pathlib but good to know)
os.path.join("data", "file.txt")  # OS-safe join
os.path.exists("data.csv")         # True/False
os.path.isfile("data.csv")
os.path.isdir("data")
os.path.getsize("file.txt")      # file size in bytes

# Environment variables
home = os.environ.get("HOME", "/home/user")  # with default
print(os.environ["PATH"])

sys — System-Specific Parameters

sys module
import sys

# Command-line arguments
# python script.py arg1 arg2
print(sys.argv)            # ['script.py', 'arg1', 'arg2']
print(sys.argv[0])         # 'script.py'  — script name

# Python version info
print(sys.version)
print(sys.version_info)    # sys.version_info(major=3, minor=12, ...)

# Exit the program
if len(sys.argv) < 2:
    print("Usage: python script.py <filename>")
    sys.exit(1)              # exit with error code

# Module search path
print(sys.path)            # list of dirs Python searches for modules
sys.path.append("/my/custom/modules")  # add custom path

# Recursion limit
print(sys.getrecursionlimit())  # default 1000

Section 7.9

re — Regular Expressions

A regular expression (regex) is a pattern for matching text. Python's re module lets you search, validate, extract, and replace text based on patterns.

Core Functions

re module functions
import re

text = "My phone is 9876543210 and email is alice@example.com"

# re.search() — find first match anywhere in string
match = re.search(r"\d{10}", text)  # \d = digit, {10} = exactly 10
if match:
    print(match.group())    # 9876543210

# re.match() — match only at start of string
re.match(r"My", text)

# re.findall() — all non-overlapping matches as list
emails = re.findall(r"[\w.]+@[\w.]+\.\w+", text)
digits = re.findall(r"\d+", "a1 bb22 ccc333")
print(digits)   # ['1', '22', '333']

# re.sub() — replace matches
clean = re.sub(r"\s+", " ", "too   many   spaces")
print(clean)   # "too many spaces"

# re.split() — split by pattern
parts = re.split(r"[,;\s]+", "a, b;c  d")
print(parts)   # ['a', 'b', 'c', 'd']

# Compile for reuse (more efficient)
phone_pattern = re.compile(r"\b\d{10}\b")
print(phone_pattern.findall(text))

Common Regex Patterns

PatternMatches
\dAny digit (0–9)
\DAny non-digit
\wWord character (letter, digit, underscore)
\sAny whitespace (space, tab, newline)
.Any character except newline
^Start of string
$End of string
*0 or more repetitions
+1 or more repetitions
?0 or 1 (optional)
{n}Exactly n repetitions
{n,m}Between n and m repetitions
[abc]Any one of a, b, c
[^abc]Any character except a, b, c
(abc)Capture group
a|ba or b

Practical Validation Examples

Validation with regex
import re

def is_valid_email(email):
    pattern = r"^[\w.+\-]+@[\w\-]+\.[a-zA-Z]{2,}$"
    return bool(re.match(pattern, email))

def is_valid_phone(phone):
    pattern = r"^[6-9]\d{9}$"   # Indian mobile number
    return bool(re.match(pattern, phone))

def is_strong_password(pwd):
    # At least 8 chars, 1 upper, 1 lower, 1 digit, 1 special
    return (
        len(pwd) >= 8 and
        re.search(r"[A-Z]", pwd) and
        re.search(r"[a-z]", pwd) and
        re.search(r"\d", pwd)    and
        re.search(r"[!@#$%^&*]", pwd)
    )
